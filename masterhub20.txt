-- Função genérica: considera inimigo se times diferentes OU se não houver times
local function IsEnemy(targetPlayer)
    local myTeam = Player.Team
    local otherTeam = targetPlayer.Team
    if myTeam and otherTeam then
        return myTeam ~= otherTeam
    end
    -- Se algum não tem time, considera todos como inimigos (jogos sem equipe)
    return true
end

-- Função estrita: usada no Farm, só considera inimigo se ambos têm time e são diferentes
local function IsEnemyStrict(targetPlayer)
    local myTeam = Player.Team
    local otherTeam = targetPlayer.Team
    if myTeam and otherTeam then
        return myTeam ~= otherTeam
    end
    return false
end

local function CreateESPForPlayer(targetPlayer)
    if not DrawingLib then return end
    local esp = {}

    esp.Line = DrawingLib.new("Line")
    esp.Line.Color = Color3.fromRGB(255, 255, 0)
    esp.Line.Thickness = 2
    esp.Line.Visible = false

    esp.BoxLines = {}
    for i = 1, 4 do
        local line = DrawingLib.new("Line")
        line.Color = Color3.fromRGB(255, 0, 0)
        line.Thickness = 2
        line.Visible = false
        esp.BoxLines[i] = line
    end

    esp.HealthBg = DrawingLib.new("Line")
    esp.HealthBg.Color = Color3.fromRGB(50, 50, 50)
    esp.HealthBg.Thickness = 4
    esp.HealthBg.Visible = false
    esp.HealthFill = DrawingLib.new("Line")
    esp.HealthFill.Color = Color3.fromRGB(0, 255, 0)
    esp.HealthFill.Thickness = 4
    esp.HealthFill.Visible = false

    esp.NameText = DrawingLib.new("Text")
    esp.NameText.Text = targetPlayer.Name
    esp.NameText.Size = 14
    esp.NameText.Center = true
    esp.NameText.Outline = true
    esp.NameText.Color = Color3.fromRGB(255, 255, 255)
    esp.NameText.Visible = false

    esp.SkeletonLines = {}
    for i = 1, #Bones do
        local line = DrawingLib.new("Line")
        line.Color = Color3.fromRGB(0, 255, 255)
        line.Thickness = 1
        line.Visible = false
        esp.SkeletonLines[i] = line
    end

    ESPObjects[targetPlayer] = esp
    return esp
end

local function RemoveESPForPlayer(targetPlayer)
    local esp = ESPObjects[targetPlayer]
    if not esp then return end
    if esp.Line then esp.Line:Remove() end
    for _, l in ipairs(esp.BoxLines or {}) do if l then l:Remove() end end
    if esp.HealthBg then esp.HealthBg:Remove() end
    if esp.HealthFill then esp.HealthFill:Remove() end
    if esp.NameText then esp.NameText:Remove() end
    for _, l in ipairs(esp.SkeletonLines or {}) do if l then l:Remove() end end
    ESPObjects[targetPlayer] = nil
end

local function UpdateESP()
    if not DrawingLib then return end
    local viewportSize = Camera.ViewportSize
    local screenCenter = Vector2.new(viewportSize.X / 2, viewportSize.Y / 2)

    local rgbColor = nil
    if GetState("RGB GERAL") then
        rgbColor = Color3.fromHSV((tick() % 5) / 5, 1, 1)
    end

    for player, esp in pairs(ESPObjects) do
        local character = player.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then
            esp.Line.Visible = false
            for _, l in ipairs(esp.BoxLines) do l.Visible = false end
            esp.HealthBg.Visible = false
            esp.HealthFill.Visible = false
            esp.NameText.Visible = false
            for _, l in ipairs(esp.SkeletonLines) do l.Visible = false end
        else
            local root = character.HumanoidRootPart
            local head = character:FindFirstChild("Head")
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            if not humanoid or not root then
                esp.Line.Visible = false
                for _, l in ipairs(esp.BoxLines) do l.Visible = false end
                esp.HealthBg.Visible = false
                esp.HealthFill.Visible = false
                esp.NameText.Visible = false
                for _, l in ipairs(esp.SkeletonLines) do l.Visible = false end
            else
                local rootPos, onScreen = Camera:WorldToViewportPoint(root.Position)
                local headPos = head and Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0)) or rootPos
                local footPos = Camera:WorldToViewportPoint(root.Position - Vector3.new(0, 3, 0))

                if not onScreen then
                    esp.Line.Visible = false
                    for _, l in ipairs(esp.BoxLines) do l.Visible = false end
                    esp.HealthBg.Visible = false
                    esp.HealthFill.Visible = false
                    esp.NameText.Visible = false
                    for _, l in ipairs(esp.SkeletonLines) do l.Visible = false end
                else
                    -- Line (tracer)
                    if GetState("ESP LINE") then
                        esp.Line.From = Vector2.new(rootPos.X, rootPos.Y)
                        esp.Line.To = screenCenter
                        esp.Line.Color = rgbColor or Color3.fromRGB(255, 255, 0)
                        esp.Line.Visible = true
                    else
                        esp.Line.Visible = false
                    end

                    -- Box
                    if GetState("ESP BOX") then
                        local height = math.abs(footPos.Y - headPos.Y)
                        local width = height * 0.65
                        local x = rootPos.X - width / 2
                        local y = headPos.Y
                        local topLeft = Vector2.new(x, y)
                        local topRight = Vector2.new(x + width, y)
                        local bottomLeft = Vector2.new(x, y + height)
                        local bottomRight = Vector2.new(x + width, y + height)

                        local boxColor = rgbColor or (IsEnemy(player) and Color3.fromRGB(255, 0, 0) or Color3.fromRGB(0, 120, 255))
                        esp.BoxLines[1].From = topLeft
                        esp.BoxLines[1].To = topRight
                        esp.BoxLines[2].From = topRight
                        esp.BoxLines[2].To = bottomRight
                        esp.BoxLines[3].From = bottomRight
                        esp.BoxLines[3].To = bottomLeft
                        esp.BoxLines[4].From = bottomLeft
                        esp.BoxLines[4].To = topLeft
                        for _, l in ipairs(esp.BoxLines) do
                            l.Color = boxColor
                            l.Visible = true
                        end
                    else
                        for _, l in ipairs(esp.BoxLines) do l.Visible = false end
                    end

                    -- Nome
                    if GetState("ESP NOME") then
                        esp.NameText.Position = Vector2.new(rootPos.X, headPos.Y - 20)
                        esp.NameText.Color = rgbColor or Color3.fromRGB(255, 255, 255)
                        esp.NameText.Visible = true
                    else
                        esp.NameText.Visible = false
                    end

                    -- Barra de vida (vertical, à esquerda do box)
                    if GetState("ESP VIDA (BARRA)") then
                        local healthPercent = humanoid.Health / humanoid.MaxHealth
                        local boxHeight = math.abs(footPos.Y - headPos.Y)
                        local boxWidth = boxHeight * 0.65
                        local boxLeft = rootPos.X - boxWidth / 2
                        local barX = boxLeft - 6
                        local barTop = headPos.Y
                        local barBottom = footPos.Y

                        esp.HealthBg.From = Vector2.new(barX, barBottom)
                        esp.HealthBg.To = Vector2.new(barX, barTop)
                        esp.HealthBg.Visible = true

                        local fillY = barBottom - (barBottom - barTop) * healthPercent
                        esp.HealthFill.From = Vector2.new(barX, barBottom)
                        esp.HealthFill.To = Vector2.new(barX, fillY)
                        esp.HealthFill.Color = rgbColor or Color3.fromHSV(healthPercent * 0.33, 1, 1)
                        esp.HealthFill.Visible = true
                    else
                        esp.HealthBg.Visible = false
                        esp.HealthFill.Visible = false
                    end

                    -- Esqueleto
                    if GetState("ESP SKELETO") then
                        for i, bone in ipairs(Bones) do
                            local part1 = character:FindFirstChild(bone[1])
                            local part2 = character:FindFirstChild(bone[2])
                            if part1 and part2 then
                                local p1, on1 = Camera:WorldToViewportPoint(part1.Position)
                                local p2, on2 = Camera:WorldToViewportPoint(part2.Position)
                                if on1 and on2 then
                                    esp.SkeletonLines[i].From = Vector2.new(p1.X, p1.Y)
                                    esp.SkeletonLines[i].To = Vector2.new(p2.X, p2.Y)
                                    esp.SkeletonLines[i].Color = rgbColor or Color3.fromRGB(0, 255, 255)
                                    esp.SkeletonLines[i].Visible = true
                                else
                                    esp.SkeletonLines[i].Visible = false
                                end
                            else
                                esp.SkeletonLines[i].Visible = false
                            end
                        end
                    else
                        for _, l in ipairs(esp.SkeletonLines) do l.Visible = false end
                    end
                end
            end
        end
    end
end

local function createFOV()
    if FOVCircle then pcall(function() FOVCircle:Remove() end) end
    FOVCircle = DrawingLib.new("Circle")
    FOVCircle.Visible = false
    FOVCircle.Radius = FOVSize
    FOVCircle.Color = Color3.fromRGB(255, 255, 255) -- cor inicial, depois muda
    FOVCircle.Thickness = 2
    FOVCircle.Filled = false
    FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
end

local function AimbotFunction()
    if not AimbotEnabled then
        if FOVCircle then FOVCircle.Visible = false end
        return
    end
    if not FOVCircle then createFOV() end
    if FOVCircle then
        FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        FOVCircle.Radius = FOVSize
        FOVCircle.Visible = true
        -- Cor RGB dinâmica
        FOVCircle.Color = Color3.fromHSV((tick() % 5) / 5, 1, 1)
    end

    local closestPlayer = nil
    local shortestDist = math.huge
    for _, targetPlayer in ipairs(Players:GetPlayers()) do
        if targetPlayer ~= Player and targetPlayer.Character then
            local character = targetPlayer.Character
            local head = character:FindFirstChild("Head")
            local root = character:FindFirstChild("HumanoidRootPart")
            local targetPart = head or root
            if targetPart and IsEnemy(targetPlayer) then
                local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local distFromCenter = (Vector2.new(screenPos.X, screenPos.Y) - Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)).Magnitude
                    if distFromCenter <= FOVSize and distFromCenter < shortestDist then
                        -- Raycast para verificar visibilidade
                        local cameraPos = Camera.CFrame.Position
                        local targetPos = targetPart.Position
                        local rayParams = RaycastParams.new()
                        rayParams.FilterType = Enum.RaycastFilterType.Exclude
                        rayParams.FilterDescendantsInstances = {Player.Character}
                        local rayResult = Workspace:Raycast(cameraPos, targetPos - cameraPos, rayParams)
                        if not rayResult or (rayResult.Instance:IsDescendantOf(character)) then
                            shortestDist = distFromCenter
                            closestPlayer = targetPlayer
                        end
                    end
                end
            end
        end
    end

    if closestPlayer and closestPlayer.Character then
        local targetPart = closestPlayer.Character:FindFirstChild("Head") or closestPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetPart then
            Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, targetPart.Position)
        end
    end
end

RunService.RenderStepped:Connect(AimbotFunction)

local function FireClick()
    pcall(function()
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 1)
        task.wait(0.05)
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 1)
    end)
end

local function IsAimingAt(targetCharacter)
    if not targetCharacter or not targetCharacter:FindFirstChild("Head") then return false end
    local cameraPos = Camera.CFrame.Position
    local targetPos = targetCharacter.Head.Position
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {Player.Character}
    local rayResult = Workspace:Raycast(cameraPos, targetPos - cameraPos, rayParams)
    if rayResult then
        return rayResult.Instance:IsDescendantOf(targetCharacter)
    end
    return true
end

local function TeleportTo(targetCharacter)
    if not targetCharacter or not targetCharacter:FindFirstChild("HumanoidRootPart") then return end
    local root = targetCharacter.HumanoidRootPart
    local targetPos = root.Position + Vector3.new(0, 3, 0)
    local myChar = Player.Character
    if myChar and myChar:FindFirstChild("HumanoidRootPart") then
        myChar:SetPrimaryPartCFrame(CFrame.new(targetPos))
    end
end

local function FarmLoop()
    while GetState("Farm Automático") do
        task.wait(0.2)

        local closestEnemy = nil
        local shortestDistance = math.huge

        -- Encontrar inimigo mais próximo (estrito)
        for _, targetPlayer in ipairs(Players:GetPlayers()) do
            if targetPlayer ~= Player and IsEnemyStrict(targetPlayer) then
                local character = targetPlayer.Character
                if character and character.Parent == Workspace then
                    local root = character:FindFirstChild("HumanoidRootPart")
                    local humanoid = character:FindFirstChildOfClass("Humanoid")
                    if root and humanoid and humanoid.Health > 0 then
                        local myRoot = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
                        if myRoot then
                            local distance = (root.Position - myRoot.Position).Magnitude
                            if distance < shortestDistance then
                                shortestDistance = distance
                                closestEnemy = targetPlayer
                            end
                        end
                    end
                end
            end
        end

        if closestEnemy and closestEnemy.Character and closestEnemy.Character:FindFirstChild("HumanoidRootPart") then
            local targetCharacter = closestEnemy.Character
            local targetHumanoid = targetCharacter:FindFirstChildOfClass("Humanoid")

            if targetHumanoid and targetHumanoid.Health > 0 then
                if GetState("Teleporte para Inimigo") then
                    TeleportTo(targetCharacter)
                    task.wait(0.1)
                end

                if GetState("Click Automático") then
                    local rootPos = targetCharacter.HumanoidRootPart.Position
                    Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, rootPos)
                    if IsAimingAt(targetCharacter) then
                        FireClick()
                    end
                else
                    local rootPos = targetCharacter.HumanoidRootPart.Position
                    Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, rootPos)
                end

                local killTimeout = 0
                while GetState("Farm Automático") and targetHumanoid.Health > 0 and killTimeout < 5 do
                    task.wait(0.1)
                    killTimeout = killTimeout + 0.1
                end

                task.wait(0.5)
            end
        end
    end
end

local function StartFarmIfNeeded()
    if FarmConnection then
        FarmConnection:Disconnect()
        FarmConnection = nil
    end
    if GetState("Farm Automático") then
        FarmConnection = RunService.Heartbeat:Connect(FarmLoop)
    end
end

local ToggleActions = {}

local function ApplyToggle(name, enabled)
    if name == "Master Mode" then
        local char = Player.Character
        if char and char:FindFirstChildOfClass("Humanoid") then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if enabled then
                hum.MaxHealth = math.huge
                hum.Health = math.huge
            else
                hum.MaxHealth = 100
                hum.Health = 100
            end
        end
    elseif name == "Performance" then
        if enabled then
            Lighting.GlobalShadows = false
            Lighting.FogEnd = 100
            if Workspace:FindFirstChildOfClass("Terrain") then
                Workspace:FindFirstChildOfClass("Terrain").WaterWaveSize = 0
            end
            pcall(function() settings().Rendering.QualityLevel = 1 end)
        else
            Lighting.GlobalShadows = true
            Lighting.FogEnd = 100000
            if Workspace:FindFirstChildOfClass("Terrain") then
                Workspace:FindFirstChildOfClass("Terrain").WaterWaveSize = 0.2
            end
            pcall(function() settings().Rendering.QualityLevel = 10 end)
        end
    elseif name == "Notifications" then
        -- feedback
    elseif name == "ESP LINE" or name == "ESP BOX" or name == "ESP VIDA (BARRA)" or name == "ESP SKELETO" or name == "ESP NOME" or name == "RGB GERAL" then
        -- Apenas altera estado; UpdateESP renderiza
    elseif name == "RAIO-X" then
        ApplyXray(enabled)
    elseif name == "Aim Preview" then
        AimbotEnabled = enabled
        if not enabled and FOVCircle then FOVCircle.Visible = false end
    elseif name == "Crosshair Preview" then
        -- opcional
    elseif name == "Speed Preview" then
        local char = Player.Character

local function CreateToggle(Name, Y)
    local Button = Instance.new("TextButton")
    Button.Parent = Content
    Button.Size = UDim2.new(1, -20, 0, 35)
    Button.Position = UDim2.fromOffset(10, Y)
    Instance.new("UICorner", Button).CornerRadius = UDim.new(0, 8)

    local function UpdateButton()
        local Enabled = GetState(Name)
        if Enabled then
            Button.BackgroundColor3 = Color3.fromRGB(20, 110, 230)
            Button.Text = Name .. "    [ ON ]"
        else
            Button.BackgroundColor3 = Color3.fromRGB(25, 29, 42)
            Button.Text = Name .. "    [ OFF ]"
        end
        Button.TextColor3 = Color3.new(1, 1, 1)
        Button.TextSize = 13
        Button.Font = Enum.Font.GothamMedium
    end
    UpdateButton()

    Button.MouseButton1Click:Connect(function()
        local newState = ChangeState(Name)
        UpdateButton()
        if ToggleActions[Name] then
            ToggleActions[Name](Name, newState)
        end
    end)
end

local function CreateFOVSlider()
    local label = Instance.new("TextLabel")
    label.Parent = Content
    label.Size = UDim2.new(1, -20, 0, 25)
    label.Position = UDim2.fromOffset(10, 120)
    label.BackgroundTransparency = 1
    label.Text = "Tamanho do FOV: " .. FOVSize
    label.TextColor3 = Color3.fromRGB(220, 220, 230)
    label.TextSize = 13
    label.Font = Enum.Font.GothamMedium
    label.TextXAlignment = Enum.TextXAlignment.Left

    local slider = Instance.new("TextButton")
    slider.Parent = Content
    slider.Size = UDim2.new(1, -20, 0, 30)
    slider.Position = UDim2.fromOffset(10, 150)
    slider.BackgroundColor3 = Color3.fromRGB(40, 44, 60)
    slider.Text = ""
    Instance.new("UICorner", slider).CornerRadius = UDim.new(0, 6)

    local function UpdateFromInput(input)
        local relX = (input.Position.X - slider.AbsolutePosition.X) / slider.AbsoluteSize.X
        FOVSize = math.clamp(math.floor(relX * 500), 50, 500)
        label.Text = "Tamanho do FOV: " .. FOVSize
        if FOVCircle then FOVCircle.Radius = FOVSize end
    end

    slider.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            UpdateFromInput(input)
        end
    end)
    slider.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
                UpdateFromInput(input)
            end
        end
    end)
end

local Dragging = false
local DragStart, StartPosition = nil, nil

Title.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        Dragging = true
        DragStart = input.Position
        StartPosition = Panel.Position
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        Dragging = false
    end
end)

UIS.InputChanged:Connect(function(input)
    if Dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - DragStart
        Panel.Position = UDim2.new(
            StartPosition.X.Scale,
            StartPosition.X.Offset + delta.X,
            StartPosition.Y.Scale,
            StartPosition.Y.Offset + delta.Y
        )
    end
end)
